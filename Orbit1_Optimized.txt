local function makeDraggable(frame, dragHandle)
    local dragging, dragInput, dragStart, startPos
    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    dragHandle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function releaseParts()
    for part, oldGroup in pairs(trackedParts) do
        if part and part:IsA("BasePart") then
            pcall(function()
                part.AssemblyLinearVelocity = Vector3.new(0, -5, 0)
                part.AssemblyAngularVelocity = Vector3.zero
                if sethiddenproperty then
                    sethiddenproperty(part, "CollisionGroupId", oldGroup)
                end
            end)
        end
    end
    table.clear(trackedParts)
    table.clear(antiFlingApplied)
end

local function isOrbitCandidate(obj)
    if not obj or not obj:IsA("BasePart") or obj.Anchored then
        return false
    end

    local character = LocalPlayer.Character
    if character and obj:IsDescendantOf(character) then
        return false
    end

    local model = obj:FindFirstAncestorOfClass("Model")
    if model and (model:FindFirstChildOfClass("Humanoid") or model:FindFirstChild("HumanoidRootPart")) then
        return false
    end

    local assemblyRoot = obj.AssemblyRootPart
    return assemblyRoot ~= nil and not assemblyRoot.Anchored
end

-- Track additions/removals instead of rescanning the entire Workspace repeatedly.
for _, obj in ipairs(workspace:GetDescendants()) do
    if isOrbitCandidate(obj) then
        candidateParts[obj] = true
    end
end

workspace.DescendantAdded:Connect(function(obj)
    if isOrbitCandidate(obj) then
        candidateParts[obj] = true
    end
end)

workspace.DescendantRemoving:Connect(function(obj)
    candidateParts[obj] = nil
    trackedParts[obj] = nil
    networkAttempted[obj] = nil
    antiFlingApplied[obj] = nil
end)

local function getOrbitRoot()
    local targetCharacter = LocalPlayer.Character
    if Config.TargetMode and Config.SelectedTarget then
        local targetPlayer = Config.SelectedTarget
        if targetPlayer and targetPlayer.Character then
            targetCharacter = targetPlayer.Character
        end
    end
    return targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
end

-- Refresh the nearby list a few times per second. New parts are event-tracked.
task.spawn(function()
    while ScreenGui and ScreenGui.Parent do
        task.wait(0.4)

        if Config.Active then
            local rootPart = getOrbitRoot()
            if rootPart then
                local nearby = {}
                local origin = rootPart.Position
                local maxDistanceSquared = Config.MaxDistance * Config.MaxDistance

                for part in pairs(candidateParts) do
                    if part and part.Parent and isOrbitCandidate(part) then
                        local offset = part.Position - origin
                        if offset:Dot(offset) <= maxDistanceSquared then
                            nearby[#nearby + 1] = part

                            -- Attempt ownership only once per part, not on every refresh.
                            if not networkAttempted[part] then
                                networkAttempted[part] = true
                                if sethiddenproperty then
                                    pcall(sethiddenproperty, part, "NetworkIsSleeping", false)
                                end
                                if part.SetNetworkOwner then
                                    pcall(function()
                                        part:SetNetworkOwner(LocalPlayer)
                                    end)
                                end
                            end
                        end
                    end
                end

                validPhysicsObjects = nearby
            end
        else
            table.clear(validPhysicsObjects)
        end
    end
end)

-- Cap the physics-update work to 30 updates per second rather than every frame.
local updateAccumulator = 0
local UPDATE_INTERVAL = 1 / 30

RunService.Heartbeat:Connect(function(deltaTime)
    if not Config.Active then
        updateAccumulator = 0
        return
    end

    updateAccumulator += deltaTime
    if updateAccumulator < UPDATE_INTERVAL then
        return
    end
    local step = updateAccumulator
    updateAccumulator = 0

    local rootPart = getOrbitRoot()
    if not rootPart then return end

    angleTime += Config.Speed * step
    local rotationCFrame = CFrame.fromEulerAnglesXYZ(Config.AngleX, Config.AngleY, Config.AngleZ)

    local basePivot = rootPart.Position + Vector3.new(Config.OffsetX, Config.OffsetY, Config.OffsetZ)
    if Config.CustomPivot then
        basePivot = Config.PivotPos + Vector3.new(Config.OffsetX, Config.OffsetY, Config.OffsetZ)
    end

    local totalParts = #validPhysicsObjects
    if totalParts == 0 then return end

    local characterCFrame = rootPart.CFrame
    local diceSideSize = math.ceil(math.sqrt(totalParts))
    local roofLayers = math.max(1, math.ceil(totalParts / 6))
    local roofRadiusScale = Config.Radius / roofLayers
    local wallLeftShift = (4 * 4) / 2

    for index = 1, totalParts do
        local part = validPhysicsObjects[index]
        if not part or not part.Parent or part.Anchored then
            continue
        end

        local targetPosition

        if Config.DiceCombatActive then
            local col = (index - 1) % diceSideSize
            local row = math.floor((index - 1) / diceSideSize)
            local localOffset = Vector3.new(
                (col - (diceSideSize - 1) / 2) * 4,
                0,
                (row - (diceSideSize - 1) / 2) * 4
            )
            targetPosition = basePivot + characterCFrame:VectorToWorldSpace(localOffset)

        elseif Config.Type == "Ring" then
            local sliceAngle = angleTime + (index * (math.pi * 2 / totalParts))
            local localOffset = Vector3.new(
                math.cos(sliceAngle) * Config.Radius,
                0,
                math.sin(sliceAngle) * Config.Radius
            )
            targetPosition = basePivot + rotationCFrame:VectorToWorldSpace(localOffset)

        elseif Config.Type == "Roof" then
            local layer = math.ceil(index / 6)
            local itemInLayer = (index - 1) % 6
            local sliceAngle = angleTime + (itemInLayer * (math.pi * 2 / 6))
            local dynamicRadius = layer * roofRadiusScale
            local localOffset = Vector3.new(
                math.cos(sliceAngle) * dynamicRadius,
                layer * 0.1,
                math.sin(sliceAngle) * dynamicRadius
            )
            targetPosition = basePivot + rotationCFrame:VectorToWorldSpace(localOffset)

        elseif Config.Type == "Wall" then
            local partsPerColumn = 5
            local columnIndex = (index - 1) % partsPerColumn
            local rowIndex = math.floor((index - 1) / partsPerColumn)
            local localRightDistance = (columnIndex * 4) - wallLeftShift
            local localOffset = Vector3.new(
                localRightDistance,
                (rowIndex * 4) - 4,
                -Config.Radius
            )
            targetPosition = basePivot + characterCFrame:VectorToWorldSpace(localOffset)

        else
            local sliceAngle = angleTime + (index * (math.pi * 2 / totalParts))
            targetPosition = basePivot + Vector3.new(
                math.cos(sliceAngle) * Config.Radius,
                0,
                math.sin(sliceAngle) * Config.Radius
            )
        end

        -- Cache the original collision group once; only update when the toggle changes.
        if sethiddenproperty then
            if trackedParts[part] == nil then
                local oldGroup = 0
                if gethiddenproperty then
                    local ok, result = pcall(gethiddenproperty, part, "CollisionGroupId")
                    if ok and result ~= nil then
                        oldGroup = result
                    end
                end
                trackedParts[part] = oldGroup
            end

            if antiFlingApplied[part] ~= Config.AntiFling then
                local groupToApply = Config.AntiFling and ORBIT_GROUP or trackedParts[part]
                pcall(sethiddenproperty, part, "CollisionGroupId", groupToApply)
                antiFlingApplied[part] = Config.AntiFling
            end
        else
            trackedParts[part] = 0
        end

        local direction = targetPosition - part.Position
        local distance = direction.Magnitude
        local targetVelocity = Vector3.zero

        if distance >= 0.5 then
            local speedScale = math.clamp(distance * 12, 10, 200)
            targetVelocity = direction * (speedScale / distance)
        end

        local currentVelocity = part.AssemblyLinearVelocity
        part.AssemblyLinearVelocity = currentVelocity + ((targetVelocity - currentVelocity) * 0.45)

        if Config.DiceCombatActive or (Config.Type ~= "Ring" and Config.Type ~= "Roof") then
            part.AssemblyAngularVelocity = Vector3.zero
        else
            part.AssemblyAngularVelocity = Vector3.new(0, 12, 0)
        end
    end
end)
