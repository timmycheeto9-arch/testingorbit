local function makeDraggable(frame, dragHandle)
    local dragging, dragInput, dragStart, startPos
    dragHandle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    dragHandle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

-- Build the candidate cache once, then maintain it with Workspace events.
-- This avoids calling GetDescendants() repeatedly while orbit is active.
for _, obj in ipairs(workspace:GetDescendants()) do
    if obj:IsA("BasePart") then
        candidateParts[obj] = true
    end
end

scanConnectionAdded = workspace.DescendantAdded:Connect(function(obj)
    if obj:IsA("BasePart") then
        candidateParts[obj] = true
    end
end)

scanConnectionRemoving = workspace.DescendantRemoving:Connect(function(obj)
    if obj:IsA("BasePart") then
        candidateParts[obj] = nil
    end
end)

task.spawn(function()
    while ScreenGui.Parent do
        task.wait(0.5)
        if Config.Active then
            local targetCharacter = LocalPlayer.Character
            if Config.TargetMode and Config.SelectedTarget and Config.SelectedTarget.Character then
                targetCharacter = Config.SelectedTarget.Character
            end
            local rootPart = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
            if rootPart then
                local tempObjects = {}
                local localCharacter = LocalPlayer.Character
                local targetPosition = rootPart.Position
                for obj in pairs(candidateParts) do
                    if obj and obj.Parent and not obj.Anchored then
                        local rootAssembly = obj.AssemblyRootPart
                        if rootAssembly and not rootAssembly.Anchored then
                            local insideLocalCharacter = localCharacter and obj:IsDescendantOf(localCharacter)
                            if not insideLocalCharacter then
                                local parentModel = obj:FindFirstAncestorOfClass("Model")
                                local isLiving = parentModel and (parentModel:FindFirstChildOfClass("Humanoid") or parentModel:FindFirstChild("HumanoidRootPart"))
                                if not isLiving and (obj.Position - targetPosition).Magnitude <= Config.MaxDistance then
                                    tempObjects[#tempObjects + 1] = obj
                                end
                            end
                        end
                    end
                end
                validPhysicsObjects = tempObjects
            end
        else
            table.clear(validPhysicsObjects)
        end
    end
end)

local physicsAccumulator = 0
heartbeatConnection = RunService.Heartbeat:Connect(function(frameDelta)
    if not Config.Active then return end
    physicsAccumulator = physicsAccumulator + frameDelta
    if physicsAccumulator < (1 / 30) then return end
    local deltaTime = physicsAccumulator
    physicsAccumulator = 0
    local targetCharacter = LocalPlayer.Character
    if Config.TargetMode and Config.SelectedTarget then
        local tPlayer = Config.SelectedTarget
        if tPlayer and tPlayer.Character then targetCharacter = tPlayer.Character end
    end
    local rootPart = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end

    angleTime = angleTime + (Config.Speed * deltaTime)
    local rotationCFrame = CFrame.fromEulerAnglesXYZ(Config.AngleX, Config.AngleY, Config.AngleZ)
    
    local basePivot = rootPart.Position + Vector3.new(Config.OffsetX, Config.OffsetY, Config.OffsetZ)
    if Config.CustomPivot then
        basePivot = Config.PivotPos + Vector3.new(Config.OffsetX, Config.OffsetY, Config.OffsetZ)
    end

    local totalParts = #validPhysicsObjects
    if totalParts == 0 then return end
    local characterCFrame = rootPart.CFrame

    for index = 1, totalParts do
        local part = validPhysicsObjects[index]
        if not part or not part.Parent then continue end
        local targetPosition
        
        if Config.DiceCombatActive then
            local sideSize = math.ceil(math.sqrt(totalParts))
            local col = (index - 1) % sideSize
            local row = math.floor((index - 1) / sideSize)
            local spacing = 4
            local localOffset = Vector3.new((col - (sideSize - 1) / 2) * spacing, 0, (row - (sideSize - 1) / 2) * spacing)
            targetPosition = basePivot + characterCFrame:VectorToWorldSpace(localOffset)
        elseif Config.Type == "Ring" then
            local sliceAngle = angleTime + (index * (math.pi * 2 / totalParts))
            local localOffset = Vector3.new(math.cos(sliceAngle) * Config.Radius, 0, math.sin(sliceAngle) * Config.Radius)
            targetPosition = basePivot + rotationCFrame:VectorToWorldSpace(localOffset)
        elseif Config.Type == "Roof" then
            local layer = math.ceil(index / 6)
            local itemInLayer = (index - 1) % 6
            local sliceAngle = angleTime + (itemInLayer * (math.pi * 2 / 6))
            local dynamicRadius = layer * (Config.Radius / math.max(1, math.ceil(totalParts / 6)))
            local localOffset = Vector3.new(math.cos(sliceAngle) * dynamicRadius, layer * 0.1, math.sin(sliceAngle) * dynamicRadius)
            targetPosition = basePivot + rotationCFrame:VectorToWorldSpace(localOffset)
        elseif Config.Type == "Wall" then
            local partsPerColumn = 5
            local columnIndex = (index - 1) % partsPerColumn
            local rowIndex = math.floor((index - 1) / partsPerColumn)
            local horizontalSpacing = 4
            local verticalSpacing = 4
            local leftShiftOffset = ((partsPerColumn - 1) * horizontalSpacing) / 2
            local localRightDistance = (columnIndex * horizontalSpacing) - leftShiftOffset
            local localHeightDistance = rowIndex * verticalSpacing
            local localOffset = Vector3.new(localRightDistance, localHeightDistance - 4, -Config.Radius)
            targetPosition = basePivot + characterCFrame:VectorToWorldSpace(localOffset)
        else
            local sliceAngle = angleTime + (index * (math.pi * 2 / totalParts))
            targetPosition = basePivot + Vector3.new(math.cos(sliceAngle) * Config.Radius, 0, math.sin(sliceAngle) * Config.Radius)
        end

        trackedParts[part] = true
        local currentVelocity = part.AssemblyLinearVelocity
        local directionVector = (targetPosition - part.Position)
        local distance = directionVector.Magnitude
        
        local targetVelocity = Vector3.zero
        if distance >= 0.5 then
            local speedScale = math.clamp(distance * 12, 10, 200)
            targetVelocity = directionVector.Unit * speedScale
        end

        local nextVelocity = currentVelocity + ((targetVelocity - currentVelocity) * 0.45)
        if Config.AntiFling and nextVelocity.Magnitude > 120 then
            nextVelocity = nextVelocity.Unit * 120
        end
        part.AssemblyLinearVelocity = nextVelocity
        
        if Config.DiceCombatActive then
            part.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        elseif Config.Type == "Ring" or Config.Type == "Roof" then
            part.AssemblyAngularVelocity = Vector3.new(0, 12, 0)
        else
            part.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        end
    end
end)
